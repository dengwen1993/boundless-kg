## 步骤 1：理解 Protocol

`src/domain/protocols.py`：

```python
class LLMClientProtocol(Protocol):
    async def chat(self, messages: list[dict], **kwargs) -> str:
        """普通对话。返回 LLM 文本响应。"""
        ...

    async def chat_structured(
        self,
        messages: list[dict],
        schema: type[T],
        **kwargs,
    ) -> T:
        """结构化对话。要求 LLM 返回符合 schema 的对象。"""
        ...
```

**关键**：你的实现**不需要 `: LLMClientProtocol**`，靠鸭子类型满足即可。

## 步骤 2：实现客户端

例：加一个智谱 AI 提供方。

```python src/infrastructure/llm/zhipu.py
from __future__ import annotations

import httpx

from src.domain.protocols import LLMClientProtocol


class ZhipuClient:
    """智谱 AI（GLM-4）客户端。OpenAI 兼容协议。"""

    def __init__(self, api_key: str, base_url: str = "https://open.bigmodel.cn/api/paas/v4", model: str = "glm-4-plus"):
        self._api_key = api_key
        self._base_url = base_url
        self._model = model
        self._client = httpx.AsyncClient(timeout=300)

    async def chat(self, messages: list[dict], **kwargs) -> str:
        resp = await self._client.post(
            f"{self._base_url}/chat/completions",
            headers={"Authorization": f"Bearer {self._api_key}"},
            json={
                "model": kwargs.get("model", self._model),
                "messages": messages,
                "temperature": kwargs.get("temperature", 0.7),
            },
        )
        resp.raise_for_status()
        return resp.json()["choices"][0]["message"]["content"]

    async def chat_structured(self, messages: list[dict], schema: type[T], **kwargs) -> T:
        # 简单实现：先 chat，再 JSON 解析
        import json
        from src.utils.json_repair import repair_json

        text = await self.chat(messages, **kwargs)
        # 让 LLM 输出 JSON（prompt 里要求）
        repaired = repair_json(text)
        return schema.model_validate_json(repaired)

    async def close(self):
        await self._client.aclose()
```

## 步骤 4：注册到 factory

`src/infrastructure/llm/factory.py`：

```python
from src.infrastructure.llm.zhipu import ZhipuClient


def get_llm_client(provider: str) -> AsyncLLMClient:
    if provider == "mock":
        return MockLLMClient()
    if provider == "minimax":
        return AnthropicCompatClient(...)
    if provider == "deepseek":
        return OpenAICompatClient(...)
    if provider == "openai":
        return OpenAICompatClient(...)
    if provider == "zhipu":                              # 新增
        return ZhipuClient(                              # 新增
            api_key=get_zhipu_api_key(),
            base_url=get_zhipu_base_url(),
            model=get_zhipu_model(),
        )
    raise ValueError(f"unknown provider: {provider}")
```

## 步骤 5：测试

`tests/unit/test_llm_zhipu.py`：

```python
import pytest
from src.infrastructure.llm.zhipu import ZhipuClient


@pytest.fixture
def mock_zhipu(monkeypatch):
    """Mock HTTP 响应，避免真实 API 调用。"""
    async def mock_post(self, url, **kwargs):
        class MockResp:
            def raise_for_status(self): pass
            def json(self):
                return {"choices": [{"message": {"content": "Mocked response"}}]}
        return MockResp()

    monkeypatch.setattr("httpx.AsyncClient.post", mock_post)


async def test_chat_returns_content(mock_zhipu):
    client = ZhipuClient(api_key="sk-test")
    result = await client.chat([{"role": "user", "content": "Hi"}])
    assert result == "Mocked response"


async def test_chat_structured_returns_model(mock_zhipu, monkeypatch):
    from pydantic import BaseModel

    class Foo(BaseModel):
        bar: str

    # mock LLM 输出 JSON
    async def mock_chat(self, messages, **kwargs):
        return '{"bar": "baz"}'

    monkeypatch.setattr(ZhipuClient, "chat", mock_chat)

    client = ZhipuClient(api_key="sk-test")
    result = await client.chat_structured([], schema=Foo)
    assert result.bar == "baz"
```

## 模式总结

| 你的 LLM 协议 | 怎么做 |
| --- | --- |
| OpenAI 兼容 | 继承 `OpenAICompatClient`，改 base_url |
| Anthropic 兼容 | 继承 `AnthropicCompatClient` |
| 自定义协议 | 写自己的 `chat` / `chat_structured` |
| 流式响应 | 额外实现 `chat_stream`（可选） |
